def stable_unique(values):
    seen = set()
    result = []
    for value in values:
        if value not in seen:
            seen.add(value)
            result.append(value)
    return result

def test_stable_unique():
    assert stable_unique([7, 2, 7, 1, 2, 9]) == [7, 2, 1, 9]
    assert stable_unique([]) == []
    assert stable_unique([4, 4, 4]) == [4]
